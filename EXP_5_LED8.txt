//Components LED-RED 8 ,PIC18F4550
#include <p18f4550.h>

void delay(unsigned int time)
{
    unsigned int i, j;
    for(i = 0; i < time; i++)
        for(j = 0; j < 155; j++);
}

void main(void)
{
    TRISD = 0x00;      // PORTD as Output
    LATD = 0x00;

    while(1)
    {
        LATD = 0xFF;
        delay(200);

        LATD = 0x00;
        delay(200);
    }
}
