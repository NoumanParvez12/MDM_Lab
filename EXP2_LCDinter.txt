//components LM016L , POT, PIC18F4550

#include <p18f4550.h>

// LCD data pins connected to PORTB
#define LCD PORTB

// LCD control pins
#define RS LATDbits.LATD0
#define EN LATDbits.LATD1


// =====================================================
// DELAY FUNCTION
// =====================================================
void delay(unsigned int t)
{
    unsigned int i, j;

    for(i = 0; i < t; i++)
    {
        for(j = 0; j < 5000; j++);
    }
}


// =====================================================
// LCD COMMAND FUNCTION
// =====================================================
void lcd_cmd(unsigned char cmd)
{
    RS = 0;          // RS = 0 means command
    LCD = cmd;       // Send command to LCD data pins

    EN = 1;          // Enable LCD
    delay(2);
    EN = 0;          // Disable LCD
}


// =====================================================
// LCD DATA FUNCTION
// =====================================================
void lcd_data(unsigned char data)
{
    RS = 1;          // RS = 1 means data
    LCD = data;      // Send character to LCD

    EN = 1;          // Enable LCD
    delay(2);
    EN = 0;          // Disable LCD
}


// =====================================================
// LCD STRING FUNCTION
// =====================================================
void lcd_string(char *str)
{
    while(*str)
    {
        lcd_data(*str++);
    }
}


// =====================================================
// MAIN FUNCTION
// =====================================================
void main()
{
    // Configure all pins as digital
    ADCON1 = 0x0F;

    // PORTB as output for LCD data
    TRISB = 0x00;

    // RD0 and RD1 as output for RS and EN
    TRISD = 0x00;

    // Initial delay after power ON
    delay(20);


    // =================================================
    // LCD INITIALIZATION
    // =================================================

    lcd_cmd(0x38);       // 8-bit mode, 2 lines, 5x7 font

    lcd_cmd(0x0C);       // Display ON, Cursor OFF

    lcd_cmd(0x06);       // Cursor moves right

    lcd_cmd(0x01);       // Clear LCD

    delay(2);


    // =================================================
    // DISPLAY MESSAGE
    // =================================================

    lcd_cmd(0x84);       // Move cursor to first line

    lcd_string("Welcome");

    delay(2);

    lcd_cmd(0xC0);       // Move cursor to second line

    lcd_string("TY MDM Students");


    // =================================================
    // INFINITE LOOP
    // =================================================

    while(1)
    {
        // Keep LCD displaying the message
    }
}
